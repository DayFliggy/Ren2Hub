func restoredControlDB(t *testing.T, models ...any) *gorm.DB {
 t.Helper()
 db, err := gorm.Open(sqlite.Open(filepath.Join(t.TempDir(), "probe.db")), &gorm.Config{Logger:glog.Default.LogMode(glog.Silent)})
 require.NoError(t,err);require.NoError(t,db.AutoMigrate(models...))
 oldDB,oldLog,oldMain,oldLogType,oldMap,oldMemory:=DB,LOG_DB,common.MainDatabaseType(),common.LogDatabaseType(),common.OptionMap,common.MemoryCacheEnabled
 DB,LOG_DB=db,db; common.SetDatabaseTypes(common.DatabaseTypeSQLite,common.DatabaseTypeSQLite);common.OptionMap=map[string]string{};common.MemoryCacheEnabled=false
 t.Cleanup(func(){DB,LOG_DB=oldDB,oldLog;common.SetDatabaseTypes(oldMain,oldLogType);common.OptionMap,common.MemoryCacheEnabled=oldMap,oldMemory;sqlDB,_:=db.DB();_ =sqlDB.Close()})
 return db
}

}

}
